━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

You ran:
  adb logcat -v threadtime com.a42r.pevplugshare:D *:S | grep -E "ADD FAVORITE|COMPLETE"

But no logs appeared. This could mean:

  1. ❓ The sync service wasn't triggered (no OsmandBackgroundSync logs)
  2. ❓ The location wasn't found in database
  3. ❓ smartSync() wasn't called
  4. ❓ addFavorite() wasn't called (updateFavorite might be called instead)
  5. ❓ Logs are there but filtered out

UNDERSTANDING THE LOG FLOW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Location Creation:
  User creates location in EditLocationActivity
  ↓
  Location saved to database (EditLocationViewModel)
  ↓
  MainActivity shows sync dialog (optional, based on settings)
  ↓
  User taps "Sync to OsmAnd"
  ↓
  EditLocationActivity starts OsmandBackgroundSyncService
  ↓
  OsmandBackgroundSyncService.onStartCommand() called
  ↓
  Service queries database for location (LocationRepository)
  ↓
  OsmandLocationSyncService.smartSync() called
  ↓
  smartSync decides: ADD or UPDATE based on location matching
  ↓
  addFavorite() or updateFavorite() called
  ↓
  API calls made to osmand.net
  ↓
  Response received
  ↓
  Database updated with OsmAnd metadata
  ↓
  ✅ COMPLETE

KEY CLASSES:
  - EditLocationActivity (.ui.edit)
  - EditLocationViewModel (.ui.edit)
  - MainActivity (.ui.main)
  - OsmandBackgroundSyncService (.services)
  - OsmandLocationSyncService (.data.repository)

TROUBLESHOOTING BY SYMPTOM
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SYMPTOM 1: No "EditLocation" logs appear
  └─ Location creation didn't complete
  └─ Check: Is location actually being saved?
  └─ Command: adb logcat | grep -i "editlocation\|save.*location"

SYMPTOM 2: Location saves but no "OsmandBackgroundSync" logs
  └─ Sync service not being started
  └─ Check: Did sync dialog appear? Did user tap it?
  └─ Check: Is startService() being called?
  └─ Command: adb logcat | grep -i "startservice\|osmandbackground"

SYMPTOM 3: Service starts but no "OsmandLocationSync" logs
  └─ Location not found in database
  └─ OR smartSync not being called
  └─ Check: Is location lookup succeeding?
  └─ Command: adb logcat | grep -i "found location\|smartsync"

SYMPTOM 4: smartSync starts but no "ADD FAVORITE" logs
  └─ addFavorite not being called
  └─ updateFavorite might be called instead
  └─ Check: Does location already exist in OsmAnd?
  └─ Command: adb logcat | grep -E "ADD FAVORITE|UPDATE FAVORITE|SMART SYNC"

SYMPTOM 5: ADD starts but doesn't complete
  └─ API call failed or timed out
  └─ Check: Network connectivity? Auth token?
  └─ Command: adb logcat | grep -E "HTTP|RESPONSE|Error|Exception"

USEFUL GREP PATTERNS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

See everything in order:
  adb logcat -c && adb logcat -v threadtime com.a42r.pevplugshare:V *:S

Just the service startup:
  adb logcat | grep -i "osmandbackgroundsync\|onstartcommand"

Just the sync process:
  adb logcat | grep -i "osmandlocationsync\|add favorite\|smartsync"

Just errors:
  adb logcat | grep -E "Error|Exception|FAILED"

Real-time with highlighting:
  adb logcat -v threadtime | grep --color=always -E "osmand|sync|error"

Save to file:
  adb logcat -v threadtime > debug_$(date +%s).log
  # Then kill with Ctrl+C after test
  grep -E "osmand|sync|error" debug_*.log

COMMON MISTAKES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ Using old APK
  → Always rebuild: ./gradlew assembleDebug -q && adb install -r ...

❌ Filter is too restrictive
  → Use :V (verbose) instead of :D (debug)
  → adb logcat com.a42r.pevplugshare:V *:S

❌ Not waiting long enough
  → API calls can take 2-5 seconds
  → Wait at least 10 seconds after tapping sync

❌ Not tapping sync button
  → Dialog must appear AND you must tap it
  → Look for "Showing sync dialog" in logs

❌ OsmAnd not connected
  → Check Settings > OsmAnd Sync > Should show "Connected as:"
  → If not connected, can't sync
